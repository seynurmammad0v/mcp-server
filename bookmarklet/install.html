IcePanel MCP — token refresh bookmarklet

Drag this button to your bookmarks bar. Then, while logged in to app.icepanel.io, click it. It will scan storage for the freshest JWT and POST it to the local refresh server.

📌 IcePanel MCP refresh
Heads up: the bookmarklet sends your JWT to http://127.0.0.1:1717. The refresh server only binds to localhost, but anyone with shell access on this machine could read the token file (~/.icepanel/bearer, mode 600). Don't run this on a shared machine without thinking.

Setup

  1. Drag the button above into your bookmarks bar.
  2. From the fork directory, run:
    node bin/refresh-server.js
  3. Open app.icepanel.io while logged in. Click the bookmarklet.
  4. The MCP server (configured with ICEPANEL_BEARER_TOKEN_FILE=~/.icepanel/bearer) will pick up the new token on the next request.

What it does

IcePanel's SPA holds the bearer JWT in JS memory only — it isn't in localStorage, sessionStorage, cookies, or IndexedDB. So the bookmarklet hooks window.fetch and XMLHttpRequest.prototype.setRequestHeader, captures the Authorization: Bearer … from the next API call the SPA makes, POSTs it to the local refresh server, then unhooks. Click around the UI to trigger an outgoing call within the 30-second window.

(function () {
  if (window.__icepanelMcpRefreshHooked) return;
  window.__icepanelMcpRefreshHooked = true;
  var TARGET = 'http://127.0.0.1:1717/token';
  var origFetch = window.fetch;
  var origXhrSetHeader = XMLHttpRequest.prototype.setRequestHeader;
  var captured = false;

  function send(token) {
    if (captured) return;
    captured = true;
    origFetch.call(window, TARGET, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ token: token })
    })
    .then(function (r) { return r.json(); })
    .then(function (j) { alert('Captured + sent token.\n\nServer: ' + JSON.stringify(j)); })
    .catch(function (e) { alert('Captured token but FAILED to reach refresh server: ' + e.message); });
  }

  function tryAuth(v) {
    if (captured || typeof v !== 'string') return;
    var m = v.match(/^Bearer (eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+)$/);
    if (m) send(m[1]);
  }

  function fromHeaders(h) {
    if (!h) return null;
    try {
      if (h instanceof Headers) return h.get('authorization');
      if (Array.isArray(h)) {
        for (var i = 0; i < h.length; i++) if (/^authorization$/i.test(h[i][0])) return h[i][1];
        return null;
      }
      return h.authorization || h.Authorization || null;
    } catch (e) { return null; }
  }

  window.fetch = function (input, init) {
    var a = fromHeaders(init && init.headers) || fromHeaders(input && input.headers);
    tryAuth(a);
    return origFetch.apply(this, arguments);
  };

  XMLHttpRequest.prototype.setRequestHeader = function (name, value) {
    if (/^authorization$/i.test(name)) tryAuth(value);
    return origXhrSetHeader.apply(this, arguments);
  };

  setTimeout(function () {
    window.fetch = origFetch;
    XMLHttpRequest.prototype.setRequestHeader = origXhrSetHeader;
    window.__icepanelMcpRefreshHooked = false;
    if (!captured) alert('No authed request seen in 30s. Click around the IcePanel UI and try again.');
  }, 30000);

  alert('Listening for outgoing IcePanel API calls. Click around the UI in the next 30s.');
})();
