IcePanel MCP — token refresh bookmarklet

Drag this button to your bookmarks bar. Then, while logged in to app.icepanel.io, click it. It will scan storage for the freshest JWT and POST it to the local refresh server.

📌 IcePanel MCP refresh
Heads up: the bookmarklet sends your JWT to http://127.0.0.1:1717. The refresh server only binds to localhost, but anyone with shell access on this machine could read the token file (~/.icepanel/bearer, mode 600). Don't run this on a shared machine without thinking.

Setup

  1. Drag the button above into your bookmarks bar.
  2. From the fork directory, run:
    node bin/refresh-server.js
  3. Open app.icepanel.io while logged in. Click the bookmarklet.
  4. The MCP server (configured with ICEPANEL_BEARER_TOKEN_FILE=~/.icepanel/bearer) will pick up the new token on the next request.

What it does

(function () {
  var found = [];
  function add(src, k, v) {
    if (typeof v !== 'string') return;
    var m = v.match(/eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+/g);
    if (!m) return;
    m.forEach(function (t) {
      try {
        var p = t.split('.')[1];
        var pad = p + '='.repeat((4 - p.length % 4) % 4);
        var d = JSON.parse(atob(pad.replace(/-/g, '+').replace(/_/g, '/')));
        if (d && d.exp) found.push({ src: src, key: k, token: t, exp: d.exp, iss: d.iss || '', aud: d.aud || '' });
      } catch (e) {}
    });
  }
  for (var i = 0; i < localStorage.length; i++) { var k = localStorage.key(i); add('localStorage', k, localStorage.getItem(k)); }
  for (var i = 0; i < sessionStorage.length; i++) { var k = sessionStorage.key(i); add('sessionStorage', k, sessionStorage.getItem(k)); }
  if (!found.length) { alert('No JWT found in storage on ' + location.host); return; }
  found.sort(function (a, b) { return b.exp - a.exp; });
  var best = found[0];
  var msg = found.length + ' JWT(s) found. Sending newest from ' + best.src + '[' + best.key + '] (iss=' + best.iss + ', exp in ' + (best.exp - Math.floor(Date.now()/1000)) + 's).';
  fetch('http://127.0.0.1:1717/token', {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ token: best.token })
  })
  .then(function (r) { return r.json(); })
  .then(function (j) { alert(msg + '\n\nServer: ' + JSON.stringify(j)); })
  .catch(function (e) { alert(msg + '\n\nFAILED to reach refresh server: ' + e.message + '\n\nIs `npm run refresh-server` running?'); });
})();
